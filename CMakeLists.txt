cmake_minimum_required(VERSION 3.20)
project(AoE2HDControl LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Este prototipo requer Windows.")
endif()
add_executable(aoe2hd_probe src/main.cpp)
target_compile_features(aoe2hd_probe PRIVATE cxx_std_17)
target_include_directories(aoe2hd_probe PRIVATE include)
target_compile_definitions(aoe2hd_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
if(MSVC)
  target_compile_options(aoe2hd_probe PRIVATE /W4 /permissive-)
endif()

enable_testing()
add_executable(aoe2hd_entity_tests tests/EntityEvidenceTests.cpp)
target_compile_features(aoe2hd_entity_tests PRIVATE cxx_std_17)
target_include_directories(aoe2hd_entity_tests PRIVATE include)
add_test(NAME entity_evidence COMMAND aoe2hd_entity_tests)
